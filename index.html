⚙️ Deployment Reference: api/health.js & api/gemini.js (Click to view/copy)

Place these two serverless functions in api/ on GitHub and set GEMINI_API_KEY in Vercel Environment Variables. Supported models: gemini-3.8-flash, gemini-2.5-flash, gemini-1.5-flash.

api/health.js
export default function handler(req, res) {
  res.status(200).json({
    ok: true,
    gemini: Boolean(process.env.GEMINI_API_KEY)
  });
}
api/gemini.js
export default async function handler(req, res) {
  if (req.method !== "POST") {
    return res.status(405).json({ error: "Method not allowed" });
  }
  const KEY = process.env.GEMINI_API_KEY;
  if (!KEY) {
    return res.status(503).json({ error: "GEMINI_API_KEY missing" });
  }
  try {
    const { parts = [], system = "", json = false } = req.body || {};
    if (!Array.isArray(parts) || parts.length === 0) {
      return res.status(400).json({ error: "Valid parts required" });
    }
    const models = (
      process.env.GEMINI_MODELS ||
      "gemini-3.8-flash,gemini-2.5-flash,gemini-1.5-flash"
    ).split(",").map((m) => m.trim()).filter(Boolean);

    let lastError = "Gemini request failed";
    for (const model of models) {
      try {
        const body = { contents: [{ role: "user", parts }] };
        if (system) body.systemInstruction = { parts: [{ text: system }] };
        if (json) body.generationConfig = { responseMimeType: "application/json" };

        const r = await fetch(
          `https://generativelanguage.googleapis.com/v1beta/models/${model}:generateContent?key=${encodeURIComponent(KEY)}`,
          {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify(body)
          }
        );
        const data = await r.json();
        if (!r.ok) throw new Error(data?.error?.message || `HTTP ${r.status}`);
        const text = data?.candidates?.[0]?.content?.parts
          ?.map((p) => p.text || "").join("\n").trim();
        if (text) return res.status(200).json({ text });
      } catch (e) {
        lastError = e.message;
      }
    }
    return res.status(502).json({ error: lastError });
  } catch (err) {
    return res.status(500).json({ error: err.message || "Server error" });
  }
}
